const num = (v: unknown): number => {
  const n = typeof v === 'string' ? parseFloat(v) : Number(v);
  return Number.isFinite(n) ? n : 0;
};

/** GET /finans/senaryo-finansal — baz vs senaryo */
export const getFinansSenaryo = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const broad = genisSantiyeRol(rol!);
    const maliyetArtis = num(req.query.maliyet_artis_pct ?? 0);
    const kurDegisim = num(req.query.kur_degisim_pct ?? 0);
    const tahsilGecikme = Math.round(num(req.query.tahsilat_gecikme_gun ?? 0));

    const sc = santiyeKapsamSql(rol!, userId, 's.id', 2);
    const sParams: unknown[] = [tenantId];
    if (sc.sql) sParams.push(sc.param);
    const sScope = broad ? '' : sc.sql;

    const sant = await query(
      `SELECT COALESCE(SUM(s.sozlesme_bedel),0)::bigint AS taahhut,
              COALESCE(SUM(s.gerceklesen),0)::bigint AS gercek,
              COALESCE(AVG(s.fiziksel_ilerleme),0)::numeric AS ort_ilerleme
       FROM santiyeler s WHERE s.tenant_id = $1 AND s.aktif = true ${sScope}`,
      sParams,
    );
    const nhParams: unknown[] = [tenantId, new Date().getFullYear()];
    let nhScope = '';
    if (!broad) {
      nhScope = ' AND n.santiye_id IN (SELECT sk.santiye_id FROM santiye_kullanicilar sk WHERE sk.kullanici_id = $3)';
      nhParams.push(userId);
    }
    const nak = await query(
      `SELECT COALESCE(SUM(tutar_try) FILTER(WHERE tip='giris'),0)::bigint AS giris,
              COALESCE(SUM(tutar_try) FILTER(WHERE tip='cikis'),0)::bigint AS cikis
       FROM nakit_hareketleri n WHERE n.tenant_id = $1 AND EXTRACT(YEAR FROM tarih)=$2 ${nhScope}`,
      nhParams,
    );
    const rowS = sant.rows[0] as Record<string, unknown>;
    const rowN = nak.rows[0] as Record<string, unknown>;
    const taahhut = num(rowS.taahhut);
    const gercek = num(rowS.gercek);
    const ortIler = num(rowS.ort_ilerleme);
    const giris = num(rowN.giris);
    const cikis = num(rowN.cikis);
    const net = giris - cikis;

    const carpanMal = 1 + maliyetArtis / 100;
    const gercekSen = Math.round(gercek * carpanMal);
    const eacBaz = ortIler > 0 ? Math.round(gercek / (ortIler / 100)) : taahhut;
    const eacSen = ortIler > 0 ? Math.round(gercekSen / (ortIler / 100)) : taahhut;

    const kurEtksi = net * (kurDegisim / 100);
    const gecikmeEtksi = -Math.round((taahhut * 0.0002) * tahsilGecikme);

    res.json({
      success: true,
      data: {
        parametreler: {
          maliyet_artis_pct: maliyetArtis,
          kur_degisim_pct: kurDegisim,
          tahsilat_gecikme_gun: tahsilGecikme,
        },
        baz: {
          taahhut_toplam: taahhut,
          gerceklesen_toplam: gercek,
          ort_fiziksel_ilerleme: ortIler,
          nakit_yil_net: net,
          eac_tahmini: eacBaz,
        },
        senaryo: {
          gerceklesen_toplam: gercekSen,
          nakit_yil_net_tahmini: Math.round(net + kurEtksi + gecikmeEtksi),
          eac_tahmini: eacSen,
          kur_etkisi_try: Math.round(kurEtksi),
          tahsil_gecikme_etkisi_try: gecikmeEtksi,
        },
        meta: {
          aciklama:
            'Senaryo sonuçları gösterge amaçlıdır; TMS/TFRS uyumu için muhasebe kuralları ile doğrulanmalıdır.',
        },
      },
    });
  } catch (e) {
    logger.error('getFinansSenaryo', e);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

/** POST /finans/tcmb-kur-sync */
export const postTcmbKurSync = async (req: Request, res: Response): Promise<void> => {
  try {
    const pack = await fetchTcmbTodayRates();
    if (!pack?.satirlar.length) {
      res.status(502).json({ success: false, message: 'TCMB kurları alınamadı (ağ veya format).' });
      return;
    }
    const tarih = pack.tarih;
    let n = 0;
    for (const s of pack.satirlar) {
      await query(
        `INSERT INTO tcmb_kurlar (tarih, kod, birim_try)
         VALUES ($1::date,$2,$3)
         ON CONFLICT (tarih, kod) DO UPDATE SET birim_try = EXCLUDED.birim_try`,
        [tarih, s.kod, s.birim_try]
      );
      n++;
    }
    res.json({ success: true, data: { tarih, eklenen: n } });
  } catch (e) {
    logger.error('postTcmbKurSync', e);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

/** POST /finans/gelir-tahakkudu — TFRS15 / TMS11 benzeri dönem geliri (fiziksel ilerleme) */
export const postGelirTahakkudu = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const donem = String((req.body as { donem?: string }).donem ?? '').trim();
    if (!/^\d{4}-\d{2}$/.test(donem)) {
      res.status(400).json({ success: false, message: 'donem YYYY-MM olmalı' });
      return;
    }

    const broad = genisSantiyeRol(rol!);
    const sc = santiyeKapsamSql(rol!, userId, 's.id', 2);
    const sp: unknown[] = [tenantId];
    if (sc.sql) sp.push(sc.param);
    const scope = broad ? '' : sc.sql;

    const rows = await query(
      `SELECT s.id, s.ad,
              COALESCE(s.sozlesme_bedel,0)::bigint AS sozlesme_bedel,
              COALESCE(s.fiziksel_ilerleme,0)::numeric AS fiziksel_ilerleme
       FROM santiyeler s
       WHERE s.tenant_id = $1 AND s.aktif = true ${scope}`,
      sp
    );

    const sonuc: unknown[] = [];
    await withTransaction(async (c) => {
      for (const raw of rows.rows) {
        const s = raw as { id: string; ad: string; sozlesme_bedel: string; fiziksel_ilerleme: string };
        if (!(await userCanAccessSantiye({ tenantId, userId, rol: rol! }, s.id))) continue;

        const bedel = num(s.sozlesme_bedel);
        const iler = num(s.fiziksel_ilerleme);
        const kumulatifHedef = Math.round((bedel * iler) / 100);

        const prev = await c.query(
          `SELECT COALESCE(MAX(kumulatif_toplam_try),0)::bigint AS k FROM gelir_tahakkuku_donem
           WHERE tenant_id = $1 AND santiye_id = $2 AND donem < $3`,
          [tenantId, s.id, donem]
        );
        const onceki = num(prev.rows[0]?.k);
        const donemGeliri = Math.max(0, kumulatifHedef - onceki);

        await c.query(
          `INSERT INTO gelir_tahakkuku_donem
            (tenant_id, santiye_id, donem, donem_geliri_try, kumulatif_toplam_try, olusturan_id)
           VALUES ($1,$2,$3,$4,$5,$6)
           ON CONFLICT (tenant_id, santiye_id, donem) DO UPDATE SET
             donem_geliri_try = EXCLUDED.donem_geliri_try,
             kumulatif_toplam_try = EXCLUDED.kumulatif_toplam_try,
             olusturan_id = EXCLUDED.olusturan_id`,
          [tenantId, s.id, donem, donemGeliri, kumulatifHedef, userId]
        );
        sonuc.push({
          santiye_id: s.id,
          ad: s.ad,
          donem_geliri_try: donemGeliri,
          kumulatif_toplam_try: kumulatifHedef,
        });
      }
    });

    res.json({ success: true, data: { donem, satirlar: sonuc } });
  } catch (e) {
    logger.error('postGelirTahakkudu', e);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

/** GET /finans/gelir-tahakkudu */

type KolonKey = 'santiye_ad' | 'sozlesme_bedel' | 'gerceklesen' | 'fiziksel_ilerleme' | 'eac' | 'sapma_yuzde';
const ALLOWED_KOLONLAR: KolonKey[] = ['santiye_ad', 'sozlesme_bedel', 'gerceklesen', 'fiziksel_ilerleme', 'eac', 'sapma_yuzde'];

const normalizeKolonlar = (raw: unknown): KolonKey[] => {
  if (!Array.isArray(raw)) return ['santiye_ad', 'sozlesme_bedel'];
  const seen = new Set<KolonKey>();
  const out: KolonKey[] = [];
  for (const k of raw) {
    const s = String(k) as KolonKey;
    if (!ALLOWED_KOLONLAR.includes(s) || seen.has(s)) continue;
    seen.add(s);
    out.push(s);
  }
  return out.length > 0 ? out : ['santiye_ad', 'sozlesme_bedel'];
};

const normalizeCron = (raw: unknown): string | null => {
  const s = String(raw ?? '').trim();
  if (!s) return null;
  const parts = s.split(/\s+/);
  if (parts.length !== 5) return null;
  return s.slice(0, 80);
};

const normalizeEmailList = (raw: unknown): string | null => {
  const s = String(raw ?? '').trim();
  if (!s) return null;
  const list = s
    .split(/[;,]/)
    .map((x) => x.trim().toLowerCase())
    .filter((x) => x.length > 3 && x.includes('@'));
  return list.length ? list.join(',') : null;
};

/** POST /finans/rapor-sablon */
export const postRaporSablon = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId } = req.user!;
    const { ad, kolonlar, zamanlama_cron, email_hedef } = req.body as {
      ad?: string;
      kolonlar?: KolonKey[];
      zamanlama_cron?: string;
      email_hedef?: string;
    };
    if (!ad?.trim()) {
      res.status(400).json({ success: false, message: 'ad zorunlu' });
      return;
    }
    const kols = normalizeKolonlar(kolonlar);

    const emails = normalizeEmailList(email_hedef);
    const ins = await query(
      `INSERT INTO rapor_sablonlari (tenant_id, ad, kolonlar, zamanlama_cron, email_hedef)
       VALUES ($1,$2,$3::jsonb,$4,$5) RETURNING id`,
      [
        tenantId,
        ad.trim(),
        JSON.stringify(kols),
        cron,
        emails,
      ]
    );
    res.status(201).json({ success: true, data: ins.rows[0] });
  } catch (e) {
    logger.error('postRaporSablon', e);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

async function raporVerisiSatirlari(tenantId: string, userId: string, rol: string): Promise<
  {
    santiye_ad: string;
    sozlesme_bedel: number;
    gerceklesen: number;
    fiziksel_ilerleme: number;
    eac: number;
    sapma_yuzde: number;
  }[]
> {
  const broad = genisSantiyeRol(rol);
  const sc = santiyeKapsamSql(rol, userId, 's.id', 2);
  const params: unknown[] = [tenantId];
  if (sc.sql) params.push(sc.param);
  const scope = broad ? '' : sc.sql;
  const rows = await query(
    `SELECT s.ad, COALESCE(s.sozlesme_bedel,0)::bigint AS sozlesme_bedel,
            COALESCE(s.gerceklesen,0)::bigint AS gerceklesen,
            COALESCE(s.fiziksel_ilerleme,0)::numeric AS fiziksel_ilerleme
     FROM santiyeler s WHERE s.tenant_id = $1 AND s.aktif = true ${scope}`,
    params
  );
  return rows.rows.map((r) => {
    const rec = r as Record<string, unknown>;
    const sozlesme = num(rec.sozlesme_bedel);
    const gercek = num(rec.gerceklesen);
    const iler = num(rec.fiziksel_ilerleme);
    const eac = iler > 0 ? Math.round(gercek / (iler / 100)) : sozlesme;
    const sapmaYuzde = sozlesme > 0 ? Number((((eac - sozlesme) / sozlesme) * 100).toFixed(1)) : 0;
    return {
      santiye_ad: String(rec.ad ?? ''),
      sozlesme_bedel: sozlesme,
      gerceklesen: gercek,
      fiziksel_ilerleme: iler,
      eac,
      sapma_yuzde: sapmaYuzde,
    };
  });
}

/** POST /finans/rapor-calistir */
export const postRaporCalistir = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const { sablon_id, format = 'json' } = req.body as { sablon_id?: string; format?: string };
    let kolonlar: KolonKey[] = ['santiye_ad', 'sozlesme_bedel', 'gerceklesen', 'fiziksel_ilerleme', 'eac', 'sapma_yuzde'];
    if (sablon_id) {
      const s = await query(`SELECT kolonlar FROM rapor_sablonlari WHERE tenant_id = $1 AND id = $2`, [
        tenantId,
        sablon_id,
      ]);
      if (s.rows[0]?.kolonlar) kolonlar = s.rows[0].kolonlar as KolonKey[];
    }

    const satirlar = await raporVerisiSatirlari(tenantId, userId, rol);
    const rows = satirlar.map((row) => {
      const o: Record<string, unknown> = {};
      for (const k of kolonlar) o[k] = row[k];
      return o;
    });

const num = (v: unknown): number => {
  const n = typeof v === 'string' ? parseFloat(v) : Number(v);
  return Number.isFinite(n) ? n : 0;
};

const tahminiKapanisTarihi = (kalanAvans: number, son90KesintiToplam: number): string | null => {
  if (kalanAvans <= 0) return new Date().toISOString().slice(0, 10);
  if (son90KesintiToplam <= 0) return null;
  const gunluk = son90KesintiToplam / 90;
  if (!Number.isFinite(gunluk) || gunluk <= 0) return null;
  const gunSayisi = Math.ceil(kalanAvans / gunluk);
  const d = new Date();
  d.setDate(d.getDate() + gunSayisi);
  return d.toISOString().slice(0, 10);
};

/**

export const getFinansSantiyeMaliyet = async (req: Request, res: Response): Promise<void> => {

    if (santiye_id && !(await userCanAccessSantiye({ tenantId, userId, rol: rol! }, santiye_id))) {

    const sc = santiyeKapsamSql(rol!, userId, 's.id', 2);

    else if (sc.sql) params.push(sc.param);

      `SELECT
         s.id, s.ad, s.il, s.durum,

         COALESCE(s.gerceklesen, 0)::bigint AS gerceklesen,
         COALESCE(s.fiziksel_ilerleme, 0)::numeric AS fiziksel_ilerleme
       FROM santiyeler s
       WHERE s.tenant_id = $1 AND s.aktif = true
         ${scope}
       ORDER BY s.ad`,
      params,
    );

      const ilerleme = num((r as Record<string, unknown>).fiziksel_ilerleme);
      const eac = ilerleme > 0 ? Math.round(gercek / (ilerleme / 100)) : sozlesme;
      const sapmaTutar = eac - sozlesme;

        a.gercek += num(rr.gerceklesen);
        a.eac += num(rr.eac);

      { sozlesme: 0, gercek: 0, eac: 0 },
    );
    const toplamSapma = toplam.eac - toplam.sozlesme;
    const toplamSapmaYuzde = toplam.sozlesme > 0 ? (toplamSapma / toplam.sozlesme) * 100 : 0;

    res.json({
      success: true,
      data: {
        satirlar: data,
        ozet: {
          sozlesme_toplam: toplam.sozlesme,
          gerceklesen_toplam: toplam.gercek,
          eac_toplam: toplam.eac,
          sapma_tutar_toplam: toplamSapma,
          sapma_yuzde_toplam: Number(toplamSapmaYuzde.toFixed(1)),
        },
        meta: {
          uyari_esik_yuzde: 10,
          kritik_esik_yuzde: 15,
        },

/**
 * Açık hakediş alacakları — planlı vadeye göre yaşlandırma (Sprint 3).
 * Planlı vade = COALESCE(ödeme, dönem bitişi, dönem başlangıcı); nakit projeksiyonu ile uyumlu.
 * Kovalar: vadesiz | vadesi_gelmedi | gun_1_30 | gun_31_60 | gun_61_90 | gun_90_plus
 */
export const getFinansAlacakYaslandirma = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const santiye_id =
      typeof req.query.santiye_id === 'string' && req.query.santiye_id.length > 0
        ? req.query.santiye_id
        : undefined;

    if (santiye_id && !(await userCanAccessSantiye({ tenantId, userId, rol: rol! }, santiye_id))) {
      res.status(403).json({ success: false, message: 'Bu şantiye için rapor görüntülenemez' });
      return;
    }

    let where = `h.tenant_id = $1 AND h.durum NOT IN ('odendi','iptal')`;
    const params: unknown[] = [tenantId];
    let idx = 2;

    if (santiye_id) {
      where += ` AND h.santiye_id = $${idx++}`;
      params.push(santiye_id);
    } else if (!genisSantiyeRol(rol!)) {
      where += ` AND h.santiye_id IN (SELECT sk.santiye_id FROM santiye_kullanicilar sk WHERE sk.kullanici_id = $${idx++})`;
      params.push(userId);
    }

    const planVade = `COALESCE(h.odeme_tarihi, h.donem_bitis, h.donem_baslangic)`;
    const kovaExpr = `CASE
      WHEN (${planVade}) IS NULL THEN 'vadesiz'
      WHEN (${planVade}) >= CURRENT_DATE THEN 'vadesi_gelmedi'
      WHEN (CURRENT_DATE - (${planVade})) BETWEEN 1 AND 30 THEN 'gun_1_30'
      WHEN (CURRENT_DATE - (${planVade})) BETWEEN 31 AND 60 THEN 'gun_31_60'
      WHEN (CURRENT_DATE - (${planVade})) BETWEEN 61 AND 90 THEN 'gun_61_90'
      ELSE 'gun_90_plus'
    END`;

    const ozet = await query(
      `SELECT ${kovaExpr} AS kova,
              COUNT(*)::int AS adet,
              COALESCE(SUM(h.toplam_tutar), 0)::bigint AS tutar
       FROM hakedisler h
       WHERE ${where}
       GROUP BY 1
       ORDER BY 1`,
      params,
    );

    const limit = Math.min(500, Math.max(20, parseInt(String(req.query.limit ?? '200'), 10) || 200));
    const paramsSatir = [...params, limit];
    const pLim = idx;
    const satirlar = await query(
      `SELECT h.id, h.no, h.toplam_tutar, h.durum, h.odeme_tarihi,
              h.donem_baslangic, h.donem_bitis,
              (${planVade}) AS vade_tarihi,
              h.santiye_id, s.ad AS santiye_adi,
              ${kovaExpr} AS kova,
              CASE
                WHEN (${planVade}) IS NOT NULL AND (${planVade}) < CURRENT_DATE
                THEN (CURRENT_DATE - (${planVade}))::int
                ELSE 0
              END AS gun_gecikme
       FROM hakedisler h
       JOIN santiyeler s ON s.id = h.santiye_id
       WHERE ${where}
       ORDER BY gun_gecikme DESC NULLS LAST, vade_tarihi ASC NULLS LAST
       LIMIT $${pLim}`,
      paramsSatir,
    );

    const rows = ozet.rows as Array<{ kova: string; adet: number; tutar: string }>;
    const tumu: Record<string, { adet: number; tutar: number }> = {};
    for (const r of rows) {
      tumu[r.kova] = { adet: r.adet, tutar: num(r.tutar) };
    }

    res.json({
      success: true,
      data: {
        referans_tarihi: new Date().toISOString().slice(0, 10),
        ozet: tumu,
        satirlar: satirlar.rows,
        meta: {
          limit,
          aciklama:
            'Kapsam: ödenmemiş ve iptal olmayan hakedişler. Planlı vade: ödeme tarihi; yoksa dönem bitişi / başlangıcı (nakit projeksiyonu ile aynı). Vadesi gelmemiş = vade bugün veya gelecek. E-posta hatırlatma ayrı entegrasyon gerektirir.',
        },
      },
    });
  } catch (error) {
    logger.error('getFinansAlacakYaslandirma hatası:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

         s.id, s.ad, s.il,
         COALESCE(s.sozlesme_bedel,0)::bigint AS sozlesme_bedel,
         COALESCE(s.gerceklesen,0)::bigint AS gerceklesen
       FROM santiyeler s
       WHERE s.tenant_id = $1
         AND s.aktif = true
         ${sScope}
       ORDER BY s.ad`,
      params,
    );

    const grupRows = await query(
      `SELECT
         COALESCE(NULLIF(TRIM(ts.is_grubu), ''), 'Tanımsız') AS is_grubu,
         COALESCE(SUM(ts.sozlesme_bedeli), 0)::bigint AS sozlesme_toplam,
         COALESCE(SUM(th.net_odeme), 0)::bigint AS odeme_toplam
       FROM taseron_sozlesmeler ts
       LEFT JOIN taseron_hakedis th ON th.sozlesme_id = ts.id AND th.tenant_id = ts.tenant_id
       WHERE ts.tenant_id = $1
         ${broad ? '' : `AND ts.santiye_id IN (SELECT sk.santiye_id FROM santiye_kullanicilar sk WHERE sk.kullanici_id = $2)`}
       GROUP BY 1
       ORDER BY 1`,
      broad ? [tenantId] : [tenantId, userId],
    );

    res.json({
      success: true,
      data: {
        santiye_merkezleri: santiyeRows.rows.map((r) => {
          const x = r as Record<string, unknown>;
          const sozlesme = num(x.sozlesme_bedel);
          const gercek = num(x.gerceklesen);
          return {
            ...x,
            kalan: sozlesme - gercek,
            kar_zarar: gercek - sozlesme,
          };
        }),
        is_grubu_dagilim: grupRows.rows.map((r) => {
          const x = r as Record<string, unknown>;
          return {
            ...x,
            acik_pozisyon: num(x.sozlesme_toplam) - num(x.odeme_toplam),
          };
        }),
      },
    });
  } catch (error) {
    logger.error('getFinansMaliyetMerkezi hatası:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }

/**
 * Aşama 3: Genel gider dağıtım anahtarı (ciro / gerçekleşen / eşit)
 */
export const getFinansGiderDagitim = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const broad = genisSantiyeRol(rol!);
    const anahtarRaw = String(req.query.anahtar ?? 'ciro').toLowerCase();
    const anahtar: 'ciro' | 'gerceklesen' | 'esit' | 'manuel' =
      anahtarRaw === 'gerceklesen' || anahtarRaw === 'esit' || anahtarRaw === 'manuel'
        ? (anahtarRaw as 'gerceklesen' | 'esit' | 'manuel')
        : 'ciro';

    const sc = santiyeKapsamSql(rol!, userId, 's.id', 2);
    const sParams: unknown[] = [tenantId];
    if (sc.sql) sParams.push(sc.param);
    const sScope = broad ? '' : sc.sql;

    const santiyeler = await query(
      `SELECT s.id, s.ad, COALESCE(s.sozlesme_bedel,0)::bigint AS sozlesme_bedel, COALESCE(s.gerceklesen,0)::bigint AS gerceklesen
       FROM santiyeler s
       WHERE s.tenant_id = $1 AND s.aktif = true ${sScope}
       ORDER BY s.ad`,
      sParams,
    );

    const giderRes = await query(
      `SELECT COALESCE(SUM(n.tutar_try),0)::bigint AS genel_gider_toplam
       FROM nakit_hareketleri n
       WHERE n.tenant_id = $1
         AND n.tip = 'cikis'
         AND n.santiye_id IS NULL`,
      [tenantId],
    );
    const toplamGider = num((giderRes.rows[0] as Record<string, unknown>)?.genel_gider_toplam);

    const rows = santiyeler.rows as Array<Record<string, unknown>>;
    let agirlikli = rows.map((r) => {
      const w =
        anahtar === 'esit' ? 1 : anahtar === 'gerceklesen' ? Math.max(0, num(r.gerceklesen)) : Math.max(0, num(r.sozlesme_bedel));
      return { ...r, w };
    });
    if (anahtar === 'manuel') {
      const m = await query(
        `SELECT santiye_id, oran_yuzde
         FROM finans_gider_dagitim_agirliklari
         WHERE tenant_id = $1`,
        [tenantId],
      );
      const bySantiye = new Map<string, number>();
      for (const row of m.rows as Array<Record<string, unknown>>) {
        bySantiye.set(String(row.santiye_id), num(row.oran_yuzde));
      }
      agirlikli = rows.map((r) => {
        const id = String(r.id ?? '');
        return { ...r, w: Math.max(0, bySantiye.get(id) ?? 0) };
      });
    }
    const wToplam = Math.max(1, agirlikli.reduce((a, r) => a + num((r as Record<string, unknown>).w), 0));

    const dagitim = agirlikli.map((r) => {
      const rx = r as Record<string, unknown>;
      const w = num((r as Record<string, unknown>).w);
      const pay = wToplam > 0 ? w / wToplam : 0;
      return {
        id: rx.id,
        ad: rx.ad,
        agirlik: Number(pay.toFixed(4)),
        oran_yuzde: Number((pay * 100).toFixed(1)),
        dagitilan_gider: Math.round(toplamGider * pay),
      };
    });

    res.json({
      success: true,
      data: {
        anahtar,
        genel_gider_toplam: toplamGider,
        dagitim,
      },
    });
  } catch (error) {
    logger.error('getFinansGiderDagitim hatası:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

export const putFinansGiderDagitimAgirlik = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId } = req.user!;
    const satirlar = Array.isArray((req.body as { satirlar?: unknown[] })?.satirlar)
      ? ((req.body as { satirlar: Array<{ santiye_id?: string; oran_yuzde?: number }> }).satirlar ?? [])
      : [];
    if (satirlar.length === 0) {
      res.status(400).json({ success: false, message: 'satirlar zorunlu' });
      return;

        res.status(403).json({ success: false, message: 'Şantiye yetkisi yok' });
        return;
      }
      toplam += num(s.oran_yuzde);
    }
    if (Math.abs(toplam - 100) > 0.5) {
      res.status(400).json({ success: false, message: 'Toplam oran %100 olmalı (±0.5)' });
      return;
    }
    for (const s of satirlar) {
      await query(
        `INSERT INTO finans_gider_dagitim_agirliklari (tenant_id, santiye_id, oran_yuzde, guncelleyen_id, guncellendi)

         DO UPDATE SET oran_yuzde = EXCLUDED.oran_yuzde, guncelleyen_id = EXCLUDED.guncelleyen_id, guncellendi = NOW()`,
        [tenantId, s.santiye_id, num(s.oran_yuzde), userId],
      );
    }
    res.json({ success: true, data: { kaydedilen: satirlar.length } });
  } catch (error) {
    logger.error('putFinansGiderDagitimAgirlik hatası:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }

/**
 * Aşama 3: Holding konsolidasyonu (tenant içi proje/il bazlı birleşik görünüm)
 */
export const getFinansHoldingKonsolidasyon = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const broad = genisSantiyeRol(rol!);
    const kapsam = String(req.query.kapsam ?? 'tenant').toLowerCase();
    let tenantIds: string[] = [tenantId];
    let holdingAd: string | null = null;

    if (kapsam === 'holding') {
      const hm = await query(
        `SELECT hg.id, hg.ad
         FROM holding_tenantleri ht
         JOIN holding_gruplari hg ON hg.id = ht.holding_id AND hg.aktif = true
         WHERE ht.tenant_id = $1
         LIMIT 1`,
        [tenantId],
      );
      const h = hm.rows[0] as { id?: string; ad?: string } | undefined;
      if (h?.id) {
        holdingAd = h.ad ?? null;
        const ts = await query(`SELECT tenant_id FROM holding_tenantleri WHERE holding_id = $1`, [h.id]);
        tenantIds = ts.rows.map((r) => String((r as { tenant_id?: string }).tenant_id ?? '')).filter(Boolean);
      }
    }
    const tenantArraySql = tenantIds.map((_, i) => `$${i + 1}`).join(',');

    const sc = santiyeKapsamSql(rol!, userId, 's.id', 2);
    const sParams: unknown[] = [...tenantIds];
    if (sc.sql && !broad) sParams.push(sc.param);
    const sScope = broad ? '' : sc.sql;

    const ilBazli = await query(
      `SELECT
         COALESCE(NULLIF(TRIM(s.il), ''), 'Tanımsız') AS il,
         COUNT(*)::int AS proje_adet,
         COALESCE(SUM(s.sozlesme_bedel),0)::bigint AS taahhut_toplam,
         COALESCE(SUM(s.gerceklesen),0)::bigint AS gerceklesen_toplam
       FROM santiyeler s
       WHERE s.tenant_id IN (${tenantArraySql}) AND s.aktif = true ${sScope}
       GROUP BY 1
       ORDER BY taahhut_toplam DESC`,
      sParams,
    );

    const genel = await query(
      `SELECT
         COUNT(*)::int AS proje_adet,
         COALESCE(SUM(s.sozlesme_bedel),0)::bigint AS taahhut_toplam,
         COALESCE(SUM(s.gerceklesen),0)::bigint AS gerceklesen_toplam
       FROM santiyeler s
       WHERE s.tenant_id IN (${tenantArraySql}) AND s.aktif = true ${sScope}`,
      sParams,
    );

    const nakit = await query(
      `SELECT
         COALESCE(SUM(n.tutar_try) FILTER (WHERE n.tip = 'giris'),0)::bigint AS giris,
         COALESCE(SUM(n.tutar_try) FILTER (WHERE n.tip = 'cikis'),0)::bigint AS cikis
       FROM nakit_hareketleri n
       WHERE n.tenant_id IN (${tenantArraySql})`,
      tenantIds,
    );

    const rowG = genel.rows[0] as Record<string, unknown>;
    const rowN = nakit.rows[0] as Record<string, unknown>;
    const giris = num(rowN.giris);
    const cikis = num(rowN.cikis);
    res.json({
      success: true,
      data: {
        kapsam: kapsam === 'holding' ? 'holding' : 'tenant',
        holding_ad: holdingAd,
        sirket_sayisi: tenantIds.length,
        proje_adet: num(rowG.proje_adet),
        taahhut_toplam: num(rowG.taahhut_toplam),
        gerceklesen_toplam: num(rowG.gerceklesen_toplam),
        net_nakit: giris - cikis,
        il_bazli: ilBazli.rows,
      },
    });
  } catch (error) {
    logger.error('getFinansHoldingKonsolidasyon hatası:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

export const listFinansTeminatlar = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const broad = genisSantiyeRol(rol!);
    const params: unknown[] = [tenantId];
    let idx = 2;
    let where = 'WHERE t.tenant_id = $1';
    const durum = typeof req.query.iade_durum === 'string' ? req.query.iade_durum : '';
    if (durum) {
      where += ` AND t.iade_durum = $${idx++}`;
      params.push(durum);
    }
    if (!broad) {
      where += ` AND (t.santiye_id IS NULL OR t.santiye_id IN (SELECT sk.santiye_id FROM santiye_kullanicilar sk WHERE sk.kullanici_id = $${idx++}))`;
      params.push(userId);
    }
    const r = await query(
      `SELECT
         t.*, s.ad AS santiye_ad, ts.sozlesme_no, tr.ad AS taseron_ad,
         (t.vade_tarihi - CURRENT_DATE)::int AS gun_kaldi
       FROM finans_teminat_kayitlari t
       LEFT JOIN santiyeler s ON s.id = t.santiye_id
       LEFT JOIN taseron_sozlesmeler ts ON ts.id = t.taseron_sozlesme_id
       LEFT JOIN taseronlar tr ON tr.id = ts.taseron_id
       ${where}
       ORDER BY t.vade_tarihi ASC`,
      params,
    );
    const ozet = await query(
      `SELECT
         COUNT(*)::int AS adet,
         COUNT(*) FILTER (WHERE vade_tarihi <= (CURRENT_DATE + INTERVAL '30 day') AND iade_durum NOT IN ('iade_edildi','reddedildi'))::int AS vade_30_gun_uyari,
         COALESCE(SUM(tutar),0)::bigint AS toplam_tutar
       FROM finans_teminat_kayitlari
       WHERE tenant_id = $1`,
      [tenantId],
    );
    res.json({ success: true, data: { satirlar: r.rows, ozet: ozet.rows[0] } });
  } catch (error) {
    logger.error('listFinansTeminatlar hatası:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

export const getFinansFiyatFarkiEskalasyon = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId, rol } = req.user!;
    const broad = genisSantiyeRol(rol!);
    const sc = santiyeKapsamSql(rol!, userId, 's.id', 2);
    const params: unknown[] = [tenantId];
    if (sc.sql) params.push(sc.param);
    const scope = broad ? '' : sc.sql;
    const k = await query(
      `SELECT
         (SELECT AVG(birim_try)::numeric FROM tcmb_kurlar WHERE kod = 'USD' AND tarih >= (CURRENT_DATE - INTERVAL '30 day')) AS usd_yeni,
         (SELECT AVG(birim_try)::numeric FROM tcmb_kurlar WHERE kod = 'USD' AND tarih >= (CURRENT_DATE - INTERVAL '120 day') AND tarih < (CURRENT_DATE - INTERVAL '90 day')) AS usd_eski`,
    );
    const usdYeni = num((k.rows[0] as Record<string, unknown>)?.usd_yeni);
    const usdEski = num((k.rows[0] as Record<string, unknown>)?.usd_eski);
    const kurDegisim = usdEski > 0 ? (usdYeni - usdEski) / usdEski : 0;
    const oranOverride = typeof req.query.kur_degisim_pct === 'string' ? num(req.query.kur_degisim_pct) / 100 : null;
    const oran = oranOverride ?? kurDegisim;
    const rows = await query(
      `SELECT s.id, s.ad, COALESCE(s.sozlesme_bedel,0)::bigint AS sozlesme_bedel, COALESCE(s.gerceklesen,0)::bigint AS gerceklesen
       FROM santiyeler s
       WHERE s.tenant_id = $1 AND s.aktif = true ${scope}
       ORDER BY s.ad`,
      params,
    );
    const satirlar = rows.rows.map((r) => {
      const x = r as Record<string, unknown>;
      const sozlesme = num(x.sozlesme_bedel);
      const gercek = num(x.gerceklesen);
      const kalan = Math.max(0, sozlesme - gercek);
      const etki = Math.round(kalan * oran * 0.35);
      return {
        ...x,
        kalan_taahhut: kalan,
        eskalasyon_oran_yuzde: Number((oran * 100).toFixed(2)),
        fiyat_farki_tutar: etki,
        revize_butce: sozlesme + etki,
      };
    });
    const toplam = satirlar.reduce((a, r) => {
      const rr = r as Record<string, unknown>;
      a.fiyat_farki_tutar += num(rr.fiyat_farki_tutar);
      a.revize_butce += num(rr.revize_butce);
      return a;
    }, { fiyat_farki_tutar: 0, revize_butce: 0 });
    res.json({
      success: true,
      data: {
        ozet: toplam,
        parametreler: {
          usd_eski: usdEski,
          usd_yeni: usdYeni,
          eskalasyon_oran_yuzde: Number((oran * 100).toFixed(2)),
          hesap_notu: 'Fiyat farkı = kalan taahhüt x kur değişimi x 0.35 (kalibrasyon parametresi).',
        },
        satirlar,
      },

       FROM hakedisler h

    const h = hakedis.rows[0];

    const yeniDurum = h.durum === 'taslak' ? 'onaylandi' : 'odendi';
    const odemeGun =
      (odeme_tarihi && String(odeme_tarihi).slice(0, 10)) || new Date().toISOString().split('T')[0];

    await withTransaction(async (client) => {
      const up = await client.query(
        `UPDATE taseron_hakedis SET durum = $2, odeme_tarihi = COALESCE($3::date, odeme_tarihi)
         WHERE id = $1 AND tenant_id = $4`,
        [id, yeniDurum, odeme_tarihi || null, tenantId],
      );
      if (!up.rowCount) {
        throw new Error('taseron_hakedis_guncellenemedi');
      }

      if (yeniDurum === 'odendi') {
        await client.query(
          `INSERT INTO nakit_hareketleri
            (tenant_id, santiye_id, tarih, tip, kategori, aciklama, tutar, tutar_try, kaydeden_id, onaylandi)
           VALUES ($1,$2,$3::date,'cikis','taseron',$4,$5,$5,$6,true)`,
          [
            tenantId,
            h.santiye_id,
            odemeGun,
            `${h.taseron_adi} - ${h.sozlesme_no || ''} ${h.donem_no}. Hakediş`,
            h.net_odeme,
            userId,
          ],
        );
      }
    });

    res.json({ success: true, mesaj: `Hakediş ${yeniDurum} durumuna getirildi` });
  } catch (error) {
    logger.error('onaylaHakedis:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

    const { santiye_id, ekipman_id, yil, ay } = req.query;

    let where = 'WHERE em.tenant_id = $1';
    const params: unknown[] = [tenantId];
    let idx = 2;

    if (santiye_id) { where += ` AND em.santiye_id = $${idx++}`; params.push(santiye_id); }
    if (ekipman_id) { where += ` AND em.ekipman_id = $${idx++}`; params.push(ekipman_id); }
    if (yil)        { where += ` AND EXTRACT(YEAR FROM em.tarih) = $${idx++}`; params.push(yil); }
    if (ay)         { where += ` AND EXTRACT(MONTH FROM em.tarih) = $${idx++}`; params.push(ay); }

    const result = await query(
      `SELECT em.*,
        e.ad as ekipman_adi, e.plaka, e.tip as ekipman_tip,
        s.ad as santiye_adi
       FROM ekipman_maliyet em
       JOIN ekipmanlar e ON e.id = em.ekipman_id
       LEFT JOIN santiyeler s ON s.id = em.santiye_id
       ${where}
       ORDER BY em.tarih DESC`,
      params
    );

    // Özet
    const ozet = await query(
      `SELECT
        tip,
        COUNT(*) as kayit_sayisi,
        SUM(tutar) as toplam
       FROM ekipman_maliyet em ${where}
       GROUP BY tip`,
      params
    );

    res.json({ success: true, data: result.rows, ozet: ozet.rows });
  } catch (error) {
    logger.error('getEkipmanMaliyet:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

export const getPuantajGrid = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId } = req.user!;
    const { santiye_id, yil, ay } = req.query;

    if (!santiye_id) {
      res.status(400).json({ success: false, message: 'santiye_id gerekli' });
      return;
    }

    // Şantiye bu tenant'a ait mi?
    const check = await query('SELECT id FROM santiyeler WHERE id = $1 AND tenant_id = $2', [santiye_id, tenantId]);
    if (!check.rows[0]) {
      res.status(403).json({ success: false, message: 'Erişim yok' });
      return;
    }

    const now = new Date();
    const y = parseInt(yil as string || String(now.getFullYear()));
    const m = parseInt(ay as string || String(now.getMonth() + 1));
    const gunSayisi = new Date(y, m, 0).getDate();

    // Personel listesi
    const personeller = await query(
      `SELECT p.id, p.ad, p.soyad, p.gorev, p.maas, p.maas_turu
       FROM personel p
       WHERE p.santiye_id = $1 AND p.tenant_id = $2 AND p.aktif = true
       ORDER BY p.ad, p.soyad`,
      [santiye_id, tenantId]
    );

    // O ay puantajları
    const puantajlar = await query(
      `SELECT personel_id, tarih, calisma_saat, fazla_mesai, tatil_mi, yevmiye
       FROM puantaj
       WHERE santiye_id = $1 AND tenant_id = $2
         AND EXTRACT(YEAR FROM tarih) = $3
         AND EXTRACT(MONTH FROM tarih) = $4`,
      [santiye_id, tenantId, y, m]
    );

    // Puantaj map
    const map: Record<string, Record<number, any>> = {};
    for (const p of puantajlar.rows) {
      const gun = new Date(p.tarih).getDate();
      if (!map[p.personel_id]) map[p.personel_id] = {};
      map[p.personel_id][gun] = p;
    }

    res.json({
      success: true,
      data: {
        personeller: personeller.rows,
        puantajlar: map,
        meta: { yil: y, ay: m, gun_sayisi: gunSayisi }
      }
    });
  } catch (error) {
    logger.error('getPuantajGrid:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

export const savePuantajGrid = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId } = req.user!;
    const { santiye_id, kayitlar } = req.body;
    // kayitlar: [{personel_id, tarih, calisma_saat, fazla_mesai, tatil_mi}]

    if (!santiye_id || !Array.isArray(kayitlar)) {
      res.status(400).json({ success: false, message: 'Geçersiz veri' });
      return;
    }

    const check = await query('SELECT id FROM santiyeler WHERE id = $1 AND tenant_id = $2', [santiye_id, tenantId]);
    if (!check.rows[0]) {
      res.status(403).json({ success: false, message: 'Erişim yok' });
      return;
    }

    let eklenen = 0;
    for (const k of kayitlar) {
      await query(
        `INSERT INTO puantaj (tenant_id, personel_id, santiye_id, tarih, calisma_saat, fazla_mesai, tatil_mi, yevmiye)
         VALUES ($1,$2,$3,$4,$5,$6,$7,$8)
         ON CONFLICT (personel_id, tarih, santiye_id) DO UPDATE SET
           calisma_saat = EXCLUDED.calisma_saat,
           fazla_mesai = EXCLUDED.fazla_mesai,
           tatil_mi = EXCLUDED.tatil_mi,

         k.calisma_saat ?? 8, k.fazla_mesai ?? 0, k.tatil_mi ?? false, k.yevmiye ?? 0]

    }

  } catch (error) {
    logger.error('savePuantajGrid:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }

const n = (v: unknown): number => {
  const x = typeof v === 'string' ? parseFloat(v) : Number(v);
  return Number.isFinite(x) ? x : 0;
};

export const getTaseronCariOzet = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId } = req.user!;
    const { santiye_id, taseron_id, baslangic_tarih, bitis_tarih } = req.query as Record<string, string | undefined>;
    const params: unknown[] = [tenantId];
    let idx = 2;
    let where = 'WHERE ts.tenant_id = $1';
    if (santiye_id) {
      where += ` AND ts.santiye_id = $${idx++}`;
      params.push(santiye_id);
    }
    if (taseron_id) {
      where += ` AND ts.taseron_id = $${idx++}`;
      params.push(taseron_id);
    }

    let hakedisTarihKosul = '';
    if (baslangic_tarih) {
      hakedisTarihKosul += ` AND COALESCE(th.odeme_tarihi, th.tarih) >= '${String(baslangic_tarih).replace(/'/g, '')}'::date`;
    }
    if (bitis_tarih) {
      hakedisTarihKosul += ` AND COALESCE(th.odeme_tarihi, th.tarih) <= '${String(bitis_tarih).replace(/'/g, '')}'::date`;
    }

    const result = await query(
      `SELECT
         ts.id AS sozlesme_id,
         ts.sozlesme_no,
         ts.is_tanimi,
         ts.sozlesme_bedeli,
         ts.avans_tutari,
         ts.stopaj_orani,
         ts.sgk_kesinti_oran,
         ts.durum,
         t.id AS taseron_id,
         t.ad AS taseron_adi,
         s.id AS santiye_id,
         s.ad AS santiye_adi,
         COALESCE(SUM(th.sozlesme_is), 0)::bigint AS hakedis_brut_toplam,
         COALESCE(SUM(th.avans_kesinti), 0)::bigint AS avans_kesinti_toplam,
         COALESCE(SUM(th.sgk_kesinti), 0)::bigint AS sgk_kesinti_toplam,
         COALESCE(SUM(th.stopaj_kesinti), 0)::bigint AS stopaj_kesinti_toplam,
         COALESCE(SUM(th.net_odeme), 0)::bigint AS net_hakedis_toplam,
         COALESCE(SUM(th.net_odeme) FILTER (WHERE th.durum = 'odendi'), 0)::bigint AS odenen_toplam
       FROM taseron_sozlesmeler ts
       JOIN taseronlar t ON t.id = ts.taseron_id
       JOIN santiyeler s ON s.id = ts.santiye_id
       LEFT JOIN taseron_hakedis th ON th.sozlesme_id = ts.id AND th.tenant_id = ts.tenant_id ${hakedisTarihKosul}
       ${where}
       GROUP BY ts.id, t.id, s.id
       ORDER BY s.ad, t.ad, ts.sozlesme_tarihi DESC`,
      params,
    );

    let uyWhereDate = '';
    if (baslangic_tarih) uyWhereDate += ` AND u.kayit_tarihi >= '${String(baslangic_tarih).replace(/'/g, '')}'::date`;
    if (bitis_tarih) uyWhereDate += ` AND u.kayit_tarihi <= '${String(bitis_tarih).replace(/'/g, '')}'::date`;
    const uyusmazlik = await query(
      `SELECT
         u.id, u.sozlesme_id, u.hakedis_id, u.kayit_tarihi, u.tip, u.seviye, u.aciklama, u.olusturuldu,
         k.ad AS kaydeden_ad, k.soyad AS kaydeden_soyad
       FROM taseron_uyusmazlik_gecmisi u
       LEFT JOIN kullanicilar k ON k.id = u.kaydeden_id
       WHERE u.tenant_id = $1 ${uyWhereDate}
       ORDER BY u.olusturuldu DESC
       LIMIT 200`,
      [tenantId],
    );
    const uyBySoz = new Map<string, Array<Record<string, unknown>>>();
    for (const row of uyusmazlik.rows as Array<Record<string, unknown>>) {
      const sid = String(row.sozlesme_id ?? '');
      const arr = uyBySoz.get(sid) ?? [];
      arr.push(row);
      uyBySoz.set(sid, arr);
    }

    const satirlar = (result.rows as Array<Record<string, unknown>>).map((r) => {
      const sozlesmeBedeli = n(r.sozlesme_bedeli);
      const brut = n(r.hakedis_brut_toplam);
      const net = n(r.net_hakedis_toplam);
      const odenen = n(r.odenen_toplam);
      const kesinHesapBakiye = sozlesmeBedeli - brut;
      const odemeBakiye = net - odenen;
      const uy = uyBySoz.get(String(r.sozlesme_id ?? '')) ?? [];
      return {
        ...r,
        kesin_hesap_bakiye: kesinHesapBakiye,
        odeme_bakiye: odemeBakiye,
        borc_alacak_durum: odemeBakiye > 0 ? 'borc' : 'kapali',
        uyusmazlik_sayisi: uy.length,
        uyusmazlik_gecmisi: uy.slice(0, 10),
      };
    });

    const ozet = satirlar.reduce(
      (a, r) => {
        const x = r as Record<string, unknown>;
        a.sozlesme += n(x.sozlesme_bedeli);
        a.net += n(x.net_hakedis_toplam);
        a.odenen += n(x.odenen_toplam);
        a.stopaj += n(x.stopaj_kesinti_toplam);
        a.sgk += n(x.sgk_kesinti_toplam);
        a.kesin += n(x.kesin_hesap_bakiye);
        a.bakiye += n(x.odeme_bakiye);
        a.uyusmazlik += Number(x.uyusmazlik_sayisi ?? 0);
        return a;
      },
      { sozlesme: 0, net: 0, odenen: 0, stopaj: 0, sgk: 0, kesin: 0, bakiye: 0, uyusmazlik: 0 },
    );

    res.json({
      success: true,
      data: {
        satirlar,
        ozet: {
          sozlesme_toplam: ozet.sozlesme,
          net_hakedis_toplam: ozet.net,
          odenen_toplam: ozet.odenen,
          stopaj_toplam: ozet.stopaj,
          sgk_toplam: ozet.sgk,
          kesin_hesap_bakiye_toplam: ozet.kesin,
          odeme_bakiye_toplam: ozet.bakiye,
          uyusmazlik_adet: ozet.uyusmazlik,
        },
      },
    });
  } catch (error) {
    logger.error('getTaseronCariOzet:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

export const createTaseronUyusmazlikKaydi = async (req: Request, res: Response): Promise<void> => {
  try {
    const { tenantId, userId } = req.user!;
    const { sozlesme_id, hakedis_id, tip, seviye, aciklama, kayit_tarihi } = req.body as Record<string, unknown>;
    if (!sozlesme_id || !aciklama) {
      res.status(400).json({ success: false, message: 'sozlesme_id ve aciklama zorunlu' });
      return;
    }
    const check = await query(
      'SELECT id FROM taseron_sozlesmeler WHERE id = $1 AND tenant_id = $2',
      [sozlesme_id, tenantId],
    );
    if (!check.rows[0]) {
      res.status(404).json({ success: false, message: 'Sözleşme bulunamadı' });
      return;
    }
    const insert = await query(
      `INSERT INTO taseron_uyusmazlik_gecmisi
        (tenant_id, sozlesme_id, hakedis_id, kayit_tarihi, tip, seviye, aciklama, kaydeden_id)
       VALUES ($1,$2,$3,$4,$5,$6,$7,$8)
       RETURNING *`,
      [
        tenantId,
        sozlesme_id,
        hakedis_id ?? null,
        kayit_tarihi || new Date().toISOString().slice(0, 10),
        tip || 'uyusmazlik',
        seviye || 'acik',
        aciklama,
        userId,
      ],
    );
    res.status(201).json({ success: true, data: insert.rows[0] });
  } catch (error) {
    logger.error('createTaseronUyusmazlikKaydi:', error);
    res.status(500).json({ success: false, message: 'Sunucu hatası' });
  }
};

const HesaplarTab: React.FC = () => {
  const qc = useQueryClient()
  const [showCreate, setShowCreate] = useState(false)
  const [yeniHesap, setYeniHesap] = useState({ kod: '', ad: '', tip: 'aktif', ust_hesap_id: '', aciklama: '' })

  const { data: hesaplar = [], isLoading } = useQuery({
    queryKey: ['hesap-plani'],
    queryFn: () => api.getHesapPlani(),
  })

  const seed = useMutation({
    mutationFn: api.seedHesapPlani,
    onSuccess: () => { qc.invalidateQueries({ queryKey: ['hesap-plani'] }); toast.success('Standart hesap planı yüklendi') },
    onError:   (e: any) => toast.error(e.response?.data?.message || 'Hata'),
  })

  const createHesap = useMutation({
    mutationFn: () => apiClient.post('/muhasebe/hesap-plani', yeniHesap).then(r => r.data),
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ['hesap-plani'] })
      toast.success('Hesap eklendi')
      setShowCreate(false)
      setYeniHesap({ kod: '', ad: '', tip: 'aktif', ust_hesap_id: '', aciklama: '' })
    },
    onError: (e: any) => toast.error(e.response?.data?.message || 'Hata'),
  })

  const grouped = useMemo(() => {
    const map: Record<string, Hesap[]> = {}
    hesaplar.filter((h: Hesap) => h.kod.length === 1).forEach((h: Hesap) => {
      map[h.kod] = hesaplar.filter((c: Hesap) => c.kod.startsWith(h.kod) && c.kod !== h.kod)
    })
    return hesaplar.filter((h: Hesap) => h.kod.length === 1).map((h: Hesap) => ({ grup: h, alt: map[h.kod] || [] }))
  }, [hesaplar])

  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 16 }}>
      <div style={{ display: 'flex', gap: 10, justifyContent: 'flex-end', flexWrap: 'wrap' }}>
        {hesaplar.length === 0 && (
          <button onClick={() => seed.mutate()} disabled={seed.isPending} style={btnPrimary}>
            {seed.isPending ? 'Yükleniyor...' : '📥 Standart Hesap Planını Yükle'}
          </button>
        )}
        <button onClick={() => setShowCreate(s => !s)} style={btnSt}>+ Hesap Ekle</button>
      </div>

      {showCreate && (
        <div style={{ ...cardSt, padding: 16 }}>
          <h4 style={{ fontSize: 13, fontWeight: 700, marginBottom: 12, color: '#0f172a' }}>Yeni Hesap</h4>
          <div style={{ display: 'grid', gridTemplateColumns: '120px 1fr 1fr', gap: 10 }}>
            <div>
              <label style={labelSt}>Kod</label>
              <input value={yeniHesap.kod} onChange={e => setYeniHesap(h => ({ ...h, kod: e.target.value }))} style={inputSt} placeholder="120.01" />
            </div>
            <div>
              <label style={labelSt}>Hesap Adı</label>
              <input value={yeniHesap.ad} onChange={e => setYeniHesap(h => ({ ...h, ad: e.target.value }))} style={inputSt} placeholder="Hesap adı" />
            </div>
            <div>
              <label style={labelSt}>Tip</label>
              <select value={yeniHesap.tip} onChange={e => setYeniHesap(h => ({ ...h, tip: e.target.value }))} style={selectSt}>
                <option value="aktif">Aktif</option>
                <option value="pasif">Pasif</option>
                <option value="gelir">Gelir</option>
                <option value="gider">Gider</option>
                <option value="oz_sermaye">Öz Sermaye</option>
              </select>
            </div>
          </div>
          <div style={{ display: 'flex', gap: 10, marginTop: 12, justifyContent: 'flex-end' }}>
            <button onClick={() => setShowCreate(false)} style={btnSt}>İptal</button>
            <button onClick={() => createHesap.mutate()} disabled={createHesap.isPending || !yeniHesap.kod || !yeniHesap.ad} style={btnPrimary}>
              Ekle
            </button>
          </div>
        </div>
      )}

      {isLoading ? <Spinner /> : hesaplar.length === 0 ? (
        <EmptyMsg text="Hesap planı boş. Standart planı yükleyin veya manuel ekleyin." />

            <div key={grup.id} style={cardSt}>
              <div style={{ padding: '10px 14px', display: 'flex', alignItems: 'center', gap: 10, borderBottom: alt.length ? '1px solid #e2e8f0' : 'none' }}>
                <span style={{ fontSize: 14, fontWeight: 800, color: TIP_RENKLER[grup.tip] }}>{grup.kod}</span>
                <span style={{ fontSize: 13, fontWeight: 700, color: '#0f172a' }}>{grup.ad}</span>
                <span style={{ fontSize: 10, padding: '2px 7px', borderRadius: 10, background: `${TIP_RENKLER[grup.tip]}22`, color: TIP_RENKLER[grup.tip], marginLeft: 'auto' }}>
                  {TIP_TR[grup.tip]}
                </span>
                <span style={{ fontSize: 11, color: '#4b5563' }}>{alt.length} alt hesap</span>
              </div>
                {alt.length > 0 && (
